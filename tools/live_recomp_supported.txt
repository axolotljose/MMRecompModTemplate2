# Instructions the live recompiler can actually generate, extracted from the N64Recomp source tree that
# the Android port pins (LiveRecomp @ c1a6dc93bfa5977de0ea256562058be4f1b73353, 2025-07-19).
# Regenerate with:
#   git -C <N64Recomp> show <pinned-commit>:src/operations.cpp > /tmp/operations.cpp
#   git -C <N64Recomp> show <pinned-commit>:src/recompilation.cpp > /tmp/recompilation.cpp
#   grep -hoE '(case )?InstrId::cpu_[A-Za-z0-9_]+' /tmp/operations.cpp /tmp/recompilation.cpp \
#     | sed 's/.*InstrId::cpu_//' | sort -u
# Anything outside this list makes src/recompilation.cpp print
#   "Unhandled instruction: <name>" and return false, which the mod loader reports as
#   "Failed to recompile mod". The N64's R4300i is a MIPS III CPU, so MIPS32 Release 2+
#   encodings (mul, ext, ins, seb, seh, clz, mfhc1, mthc1, ll/sc, sync) are absent by design.
